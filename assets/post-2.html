TEA与XTEA

tea的核心是delta（一般是0x9E3779B9，黄金分割数）作为加密的开始，每轮把delta累加进sum，sum再去参与轮函数，这样每轮的结果就会不同

其中tea的轮函数是把cipher1[4]与key1[4]和sum进行加减等操作后进行xor，一共两条线，第一条用cipher的前两个数据循环处理32轮，第二条用后两个数据

xtea则是在第一步时根据sum的值取key，加密强度更高

打开ida可以看到tea_encrypt和xtea_encrypt两个函数，评价为喜欢看伪码的同学有福了，这里是加密逻辑一般用+=表示，而伪码用的-=，有点绕

这里的1640531527（0x61C88647）就是

2³² − 0x9E3779B9（delta） = 0x100000000 − 0x9E3779B9 = 0x61C88647

看汇编就是正常逻辑

密文/密钥都在ida的符号表里：tea_cipher、tea_key、xtea_cipher、xtea_key，注意小端存储要从后往前读

标准tea的加密逻辑是+=这里解密用-=

（出题人的提示好用心我好爱）

#include <stdio.h>

#define uint unsigned int

void tea_decrypt(uint cipher[2], uint key[4])
{
    uint delta = -1640531527 * 32;

    for(int i = 1; i <= 32; i++)
    {
        cipher[1] -= (cipher[0] + delta) ^ (16 * cipher[0] + key[2]) ^ ((cipher[0] >> 5) + key[3]);
        cipher[0] -= (cipher[1] + delta) ^ (16 * cipher[1] + key[0]) ^ ((cipher[1] >> 5) + key[1]);
        delta += 1640531527;
    }
}

void xtea_decrypt(uint cipher[2], uint key[4])
{
    uint delta = -1640531527 * 32;

    for(int i = 1; i <= 32; i++)
    {
        cipher[1] -= (((cipher[0] >> 5) ^ (16 * cipher[0])) + cipher[0]) ^ (key[(delta >> 11) & 3] + delta);
        delta += 1640531527;
        cipher[0] -= (((cipher[1] >> 5) ^ (16 * cipher[1])) + cipher[1]) ^ (key[delta & 3] + delta);
    }
}

{

import gdb
HDR = bytes.fromhex("7f454c4602010100000000000000000002003e0001000000d0104000000000004000000000000000e04500000000000000000000400038000d00400025002400")
class Fix67(gdb.Command):
    def __init__(self): super().__init__("fix67", gdb.COMMAND_USER)
    def invoke(self, arg, _):
        addr = int(gdb.parse_and_eval(arg or "$rdi"))
        inf = gdb.selected_inferior()
        inf.write_memory(addr, HDR)
        phdr = bytes(inf.read_memory(addr + 0x40, 13 * 0x38)).replace(b"\x67", b"\0")
        inf.write_memory(addr + 0x40, phdr)
        print(f"fixed ELF header at {addr:#x}")

#include <stdio.h>

#define uint unsigned int

#define MX (((z >> 5 ^ y << 2) + (y >> 3 ^ z << 4)) ^ ((sum ^ y) + (key[(p & 3) ^ e] ^ z)))

void xxtea_decrypt(uint v[16], uint key[4])
{
    int n = 16;
    uint delta = 0x5F3759DF;
    int rounds = 6 + 52 / n;
    uint sum = delta * rounds;
    uint y = v[0], z, e, p;

    while (rounds--)
    {
        e = (sum >> 2) & 3;
        for (p = n - 1; p > 0; p--)
        {
            z = v[p - 1];
            v[p] -= MX;
            y = v[p];
        }
        z = v[n - 1];
        v[0] -= MX;
        y = v[0];
        sum -= delta;
    }
}
