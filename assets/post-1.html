import hashlib
import multiprocessing as mp
import os

TARGET = "fc29eb768c139c05c0bfcb697d9b26d194878a66451b3ab91b202e9710874a63"
SALT = "a3f8d9c2e1b4h7g6k9m2n5p8q1r4t7w"
MAX_NUM = 100_000_000

def calc(pin: str) -> str:
    first = hashlib.sha256((pin + SALT).encode()).hexdigest()
    return hashlib.sha256(first.encode()).hexdigest()

def worker(start: int, step: int, found: mp.Event, out_q: mp.Queue) -> None:
    i = start
    checked = 0
    while i < MAX_NUM and not found.is_set():
        pin = f"{i:08d}"
        if calc(pin) == TARGET:
            out_q.put(pin)
            found.set()
            return
        i += step
        checked += 1
        if start == 0 and checked % 500_000 == 0:
            print(f"[progress] checked~{checked * step}", flush=True)

def main() -> None:
    workers = os.cpu_count() or 4
    print(f"workers={workers}", flush=True)
    print(f"target={TARGET}", flush=True)
    print(f"salt={SALT}", flush=True)
    found = mp.Event()
    out_q = mp.Queue()
    procs = []

    for s in range(workers):
        p = mp.Process(target=worker, args=(s, workers, found, out_q))
        p.start()
        procs.append(p)

    for p in procs:
        p.join()

    if not out_q.empty():
        print("FOUND:", out_q.get(), flush=True)
    else:
        print("NOT FOUND", flush=True)

if __name__ == "__main__":
    mp.freeze_support()
    main()

Frida 动态拦截是在运行时 Hook ：不改 apk 文件，在应用运行的过程中，把某个函数挂钩，在函数执行前或后，读取或修改参数、返回值、内存数据

ai 分析定位

1. 沿加载链跟函数 attachBaseContext() → decrypt(...) → parseDexPayload(decryptedPayload) → initDexFromMemory(dexData)

attachBaseContext() 是 Android Application 的早期生命周期回调。应用刚启动时会先执行它，所以壳常把解密和加载逻辑放这里

decrypt(...) 是壳里 “ 把加密载荷解密成明文字节 ” 的函数。 输入是 payload.dat 的密文，输出是后续 parseDexPayload(...) 使用的明文数据

parseDexPayload(decryptedPayload) 是把 decrypt(...) 得到的明文载荷 decryptedPayload 交给 parseDexPayload 去解析、拆分出各个 dexData

initDexFromMemory(dexData) 是把单个 dexData （ DEX 字节数组）直接在内存中加载的函数，它内部用 InMemoryDexClassLoader 完成加载

initDexFromMemory 是壳里的私有方法名。 作用：把 DEX 字节数组从内存直接加载进类加载器 。

from pathlib import Path
from Crypto.Cipher import AES
from Crypto.Util.Padding import unpad

DB = Path(r"文件路径")
OUT = Path(r"文件路径")

conn = sqlite3.connect(str(DB))
conn.row_factory = sqlite3.Row
cur = conn.cursor()

# load per-user key/iv from user.config_data
user_crypto = {}
for r in cur.execute("SELECT id, config_data FROM user"):
    uid = r["id"]
    cfg = r["config_data"]
    if not cfg:
        continue
    try:
        j = json.loads(cfg)
        ek = j.get("enc_key")
        iv = j.get("enc_iv")
        if ek and iv:
            user_crypto[uid] = (base64.b64decode(ek), base64.b64decode(iv))
    except Exception:
        pass

rows = list(cur.execute("SELECT id, conversation_id, sender_id, content, type, status, create_at FROM message ORDER BY create_at"))

with OUT.open("w", newline="", encoding="utf-8-sig") as f:
    w = csv.writer(f)
    w.writerow(["id", "conversation_id", "sender_id", "type", "status", "create_at", "content_cipher", "content_plain", "decrypt_ok"])

    for r in rows:
        mid = r["id"]
        conv = r["conversation_id"]
        sid = r["sender_id"]
        ctext_b64 = r["content"] or ""
        mtype = r["type"]
        st = r["status"]
        ts = r["create_at"]

        plain = ""
        ok = 0

        keyiv = user_crypto.get(sid)
        if keyiv:
            key, iv = keyiv
            try:
                cbytes = base64.b64decode(ctext_b64)
                pt = AES.new(key, AES.MODE_CBC, iv).decrypt(cbytes)
                pt = unpad(pt, AES.block_size)
                plain = pt.decode("utf-8", errors="replace")
                ok = 1
            except Exception:
                pass

        w.writerow([mid, conv, sid, mtype, st, ts, ctext_b64, plain, ok])

print(f"users_with_crypto={len(user_crypto)}")
print(f"messages={len(rows)}")
print(f"out={OUT}")

import hashlib
from pathlib import Path
from cryptography.hazmat.primitives.ciphers.aead import AESGCM

# ===== 1) 输入文件 =====
# 这是待解密的 JTH3 文件路径
p = Path(r"文件路径")
d = p.read_bytes()  # 读取整个文件为 bytes

# ===== 2) 按逆向得到的参数派生 AES 密钥 =====
# key = PBKDF2-HMAC-SHA256(
#     password = "Jinritoutiao_Master_Key_2024_Secret",
#     salt = "JinriSalt_2024",
#     iter = 65536,
#     dkLen = 32
# )
key = hashlib.pbkdf2_hmac(
    "sha256",
    b"Jinritoutiao_Master_Key_2024_Secret",
    b"JinriSalt_2024",
    65536,
    dklen=32,
)

# AES-GCM 解密器（只初始化一次）
a = AESGCM(key)

# 存放所有“可成功解密”的候选结果
# 每条结构：(score, off, tag_off, ct_len, pt_len, plaintext_head)
hits = []

# ===== 3) 暴力探测 JTH3 封装中 iv / ciphertext / tag 的偏移 =====
# off: 假设的 iv 起始偏移（iv 长度固定 12 字节）
for off in range(0, 256):
    # 至少要能取到 12 字节 iv
    if off + 12 >= len(d):
        break

    # 取 iv（GCM 常用 12 字节 nonce）
    iv = d[off : off + 12]

    # tag_off: 假设的 tag 起始偏移（tag 长度固定 16 字节）
    # tag 至少要在 iv 后面，并且留出 16 字节
    min_tag = off + 13
    max_tag = min(len(d) - 16, len(d) - 1)

    for tag_off in range(min_tag, max_tag + 1):
        tag = d[tag_off : tag_off + 16]
        if len(tag) < 16:
            break

        # 假设密文区间为 [off+12, tag_off)
        ct = d[off + 12 : tag_off]
        if not ct:
            continue

        # GCM 解密时输入格式是 ciphertext + tag
        try:
            pt = a.decrypt(iv, ct + tag, None)
        except Exception:
            # 认证失败或参数错误，直接跳过
            continue

        # ===== 4) 对成功解密的候选做“可读性打分” =====
        score = 0

        # JSON 常见起始符
        if pt[:1] in (b"{", b"["):
            score += 5

        # 业务关键词命中加分（你的样本里常见）
        if b"wallet" in pt:
            score += 3
        if b"bank" in pt:
            score += 2
        if b"password" in pt:
            score += 2

        # 可打印字符比例（越高越像正常文本 / JSON）
        printable = sum((32 <= c < 127) or c in (9, 10, 13) for c in pt) / max(1, len(pt))
        if printable > 0.85:
            score += 2

        # 记录候选
        hits.append((score, off, tag_off, len(ct), len(pt), pt[:120]))

# ===== 5) 输出候选并选择最优 =====
hits.sort(reverse=True)  # 按 score 从高到低排序

print("hits", len(hits))
for h in hits[:10]:
    # 打印前 10 条：分数、偏移、长度、明文头
    print(h[:5], h[5])

if hits:
    # 取分数最高的一条作为最佳结构
    score, off, tag_off, ct_len, pt_len, head = hits[0]
    iv = d[off : off + 12]
    tag = d[tag_off : tag_off + 16]
    ct = d[off + 12 : tag_off]

    # 用最佳参数再解一次，输出完整明文
    pt = a.decrypt(iv, ct + tag, None)
    out = Path(r"C:\Users\jzz11\Desktop\vault_decrypted_best.bin")
    out.write_bytes(pt)

    print("best", score, "off", off, "tag_off", tag_off, "pt_len", pt_len, "out", str(out))
    print("head", head)
else:
    print("no candidates")

#!/usr/bin/env python3
# -*- coding: utf-8 -*-

import argparse
import base64
import sys

XOR_KEY = 0x5A
BASE64_TABLE = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
CUSTOM_TABLE = "ZYXWVUTSRQPONMLKJIHGFEDCBAzyxwvutsrqponmlkjihgfedcba9876543210+/"

def xor_decrypt(data: str) -> str:
    return ''.join(chr(ord(ch) ^ ((XOR_KEY + i) % 256)) for i, ch in enumerate(data))

def char_swap_decode(data: str) -> str:
    trans = str.maketrans(CUSTOM_TABLE, BASE64_TABLE)
    return data.translate(trans)

def decrypt(cipher_hex: str) -> str:
    step3 = bytes.fromhex(cipher_hex.strip()).decode("utf-8")
    step2 = char_swap_decode(step3)
    step1 = base64.b64decode(step2).decode("utf-8")
    return xor_decrypt(step1)

def main():
    parser = argparse.ArgumentParser(description="Decrypt text encrypted by encrypt_tool.py")
    parser.add_argument("cipher", nargs="?", help="hex ciphertext string")
    parser.add_argument("-i", "--input", help="ciphertext file path")
    args = parser.parse_args()

    if args.input:
        with open(args.input, "r", encoding="utf-8") as f:
            cipher_hex = f.read().strip()
    elif args.cipher:
        cipher_hex = args.cipher.strip()
    else:
        cipher_hex = input("请输入密文(hex): ").strip()

        plain = decrypt(cipher_hex)

        print(plain)
    except Exception as e:

SQL：

SELECT ur.name, t.total_flow
FROM (
SELECT w.user_id, SUM(ABS(w.`change`)) AS total_flow
FROM wallet_log w
GROUP BY w.user_id
) t
JOIN user_real ur ON ur.user_id = t.user_id
WHERE ur.name IS NOT NULL
ORDER BY t.total_flow DESC
LIMIT 10;
